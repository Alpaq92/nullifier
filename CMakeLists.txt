cmake_minimum_required(VERSION 3.21)
project(grandnullifier VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(GN_TOOL "Build the nullifier console tool" ${PROJECT_IS_TOP_LEVEL})
option(GN_TESTS "Build the tests" ${PROJECT_IS_TOP_LEVEL})

add_library(grandnullifier STATIC src/grandnullifier.c)
target_include_directories(grandnullifier PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(WIN32)
    target_link_libraries(grandnullifier PRIVATE bcrypt)
endif()
if(MSVC)
    target_compile_options(grandnullifier PRIVATE /W4)
    target_compile_definitions(grandnullifier PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(grandnullifier PRIVATE -Wall -Wextra -pedantic)
endif()

if(GN_TOOL)
    add_executable(nullifier tools/nullifier.c)
    target_link_libraries(nullifier PRIVATE grandnullifier)
    if(WIN32)
        target_link_libraries(nullifier PRIVATE shell32)
    endif()
    if(MSVC)
        target_compile_definitions(nullifier PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endif()

if(GN_TESTS)
    enable_testing()
    add_executable(grandnullifier-test tests/test.c)
    target_link_libraries(grandnullifier-test PRIVATE grandnullifier)
    if(MSVC)
        target_compile_definitions(grandnullifier-test PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME grandnullifier COMMAND grandnullifier-test WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
endif()
